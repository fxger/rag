`python
def normalize_query(text: str) -> str:
    """标准化查询文本 (去除标点、停用词)"""
    text = text.lower().strip()
    # 去除标点符号
    text = re.sub(r'[？?！!。，,、\.\s\-\_\~\\@\#\$\%\^\&\*\(\)\[\]\{\}\<\>]+', '', text)
    # 去除常见停用词
    stopwords = ['的', '了', '是', '在', '有', '吗', '呢', '吧', '啊', '么', '请', '告诉', '我']
    for word in stopwords:
        text = text.replace(word, '')
    return text
`

## 2. 三级查找策略

### 2.1 缓存获取流程
`python
async def get(self, text: str) -> Optional[list[float]]:
    """从缓存获取 embedding"""
    # Step 1: L1 精确匹配 (进程内，纳秒级)
    key = self._make_key(text)
    if key in self.l1:
        self._hits["l1_exact"] += 1
        return self.l1[key]
    
    # Step 2: L1 模糊匹配 (相似度≥85%)
    fuzzy_result = self._fuzzy_search(text)
    if fuzzy_result:
        self._hits["l1_fuzzy"] += 1
        return fuzzy_result
    
    # Step 3: L2 Redis缓存 (毫秒级)
    try:
        redis = await get_cache_redis()
        raw = await redis.get(f"{self.prefix}{key}")
        if raw is not None:
            value = json.loads(raw)
            self.l1[key] = value  # 回填L1缓存
            self._hits["l2"] += 1
            return value
    except Exception as e:
        logger.warning("L2 cache read error: %s", e)
    
    self._hits["miss"] += 1
    return None
`

async def set(self, text: str, embedding: list[float], ttl: int | None = None):
    """写入缓存"""
    key = self._make_key(text)
    
    # L1: 直接存储Python对象
    self.l1[key] = embedding
    
    # L2: Redis二进制安全序列化
    try:
        redis = await get_cache_redis()
        await redis.set(
            f"{self.prefix}{key}",

        )

### 3.2 批量检索机制
`python
async def get_batch(self, text_hashes: list[str]) -> dict[str, list[float]]:
    """批量检索缓存"""
    redis = await get_cache_redis()
    keys = [f"{self.prefix}{h}" for h in text_hashes]
    
    # 使用Redis管道批量获取
    pipeline = redis.pipeline()
    for key in keys:
        pipeline.get(key)
    results = await pipeline.execute()
    
    cached = {}
    for text_hash, raw_data in zip(text_hashes, results):
        if raw_data is not None:
            embedding = json.loads(raw_data)
            cached[text_hash] = embedding
    
    return cached
`

## 5. 线程安全重排器

### 5.1 并发控制
`python
class ThreadSafeReranker:
    def __init__(self, max_concurrent: int = 4):
        # 信号量控制并发推理数
        self._semaphore = ModelSemaphore(max_concurrent)
        
        # 结果缓存 (LRU, 256条目)
        self._cache: dict[str, tuple[float, list]] = {}
        self._cache_ttl = 300  # 5分钟
    
    async def rerank(self, query: str, documents: list[SearchResult]):
        """线程安全的重排"""
        # 检查缓存
        cache_key = self._get_cache_key(query, [doc.chunk_id for doc in documents])
        cached = self._get_from_cache(cache_key)
        if cached:
            return cached
        
        # 获取信号量 (最多4个并发推理)
        if not self._semaphore.acquire(timeout=5.0):
            # 超时降级
            return self._fallback_rerank(documents)
        
        try:
            # 执行推理
            result = await self._rerank_with_model(query, documents)
            self._set_cache(cache_key, result)
            return result
        finally:
            self._semaphore.release()
`

def _ensure_model_loaded(self) -> bool:
    """双重检查锁定确保模型只加载一次"""
    if self._model_loaded:
        return self.model is not None
    
    with self._load_lock:  # 线程锁
        if self._model_loaded:
            return self.model is not None
        
        try:
            from sentence_transformers import CrossEncoder
            self.model = CrossEncoder(self.model_name, max_length=self.max_length)
            self._model_loaded = True
            return True
        except Exception as e:
            logger.warning(f"Failed to load model: {e}")
            return False

## 6. 成本降低机制

### 6.1 缓存命中率统计
`python
@property
def stats(self) -> dict:
    """缓存命中率统计"""
    total = sum(self._hits.values())
    return {
        "l1_exact_hits": self._hits["l1_exact"],    # L1精确命中
        "l1_fuzzy_hits": self._hits["l1_fuzzy"],    # L1模糊命中
        "l2_hits": self._hits["l2"],                # L2 Redis命中
        "misses": self._hits["miss"],               # 未命中
        "hit_rate": (self._hits["l1_exact"] + 
                    self._hits["l1_fuzzy"] + 
                    self._hits["l2"]) / max(total, 1),
        "fuzzy_index_size": len(self._fuzzy_index),
    }
`
